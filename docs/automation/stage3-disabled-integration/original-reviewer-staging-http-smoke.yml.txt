name: ACA Reviewer Staging HTTP Smoke
on:
  pull_request:
    branches: [main]
    paths:
      - '.github/workflows/reviewer-staging-http-smoke.yml'
  workflow_dispatch:
permissions:
  contents: read
jobs:
  unauthenticated-surface:
    name: Staging public HTTP fail-closed smoke
    runs-on: ubuntu-latest
    timeout-minutes: 5
    steps:
      - name: Check health and unauthorized reviewer routes
        shell: bash
        run: |
          set -euo pipefail
          base='https://alphine-aca-phase2-reviewer-staging.onrender.com'
          for i in {1..8}; do
            code=$(curl --silent --show-error --max-time 12 --output /tmp/aca-health --write-out '%{http_code}' "$base/health") || code=000
            if [[ "$code" == 200 ]]; then break; fi
            sleep 8
          done
          test "$code" = 200
          node -e 'const fs=require("fs");const x=JSON.parse(fs.readFileSync("/tmp/aca-health","utf8"));if(x.status!=="up"||x.mode!=="fail_closed")process.exit(1)'
          for route in triage review; do
            code=$(curl --silent --show-error --max-time 12 --output /dev/null --write-out '%{http_code}' -X POST -H 'Content-Type: application/json' --data '{"repositoryId":1080201236,"prNumber":27,"issueNumber":27,"headSha":"0000000000000000000000000000000000000000"}' "$base/$route")
            test "$code" = 401
          done
          code=$(curl --silent --show-error --max-time 12 --output /dev/null --write-out '%{http_code}' "$base/triage")
          test "$code" = 404
          echo 'Staging health and unauthenticated fail-closed HTTP checks passed.'
